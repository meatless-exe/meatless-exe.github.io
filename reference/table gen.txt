<div id='outputText_nss'></div>

<script>

function expandString(str, map) {
	var recur_limit = 20
	var tempmap = {}

	while (str.match(/\{.*?\}/) && recur_limit > 0){
		str = str.replace(/\{(.*?)\}/g, (match) => {
			var input = match.replace(/[\{\}]/g, "").split(' -')
			var key = input[0]

			if (!map.hasOwnProperty(key))
				return key

			if (input.includes("u")) {
				var key_u = key+"_unique"
				if (!tempmap.hasOwnProperty(key_u) || tempmap[key_u].length == 0)
					tempmap[key_u] = map[key].slice()
				return tempmap[key_u].splice(Math.floor(Math.random() * tempmap[key_u].length), 1);
			}

			if (input.includes("r")) {
				var key_r = key+"_repeat"
				if (!tempmap.hasOwnProperty(key_r))
					tempmap[key_r] = map[key][Math.floor(Math.random() * map[key].length)]
				return tempmap[key_r]
			}

			if (input.includes("s")) {
				var key_r = key+"_repeat"
				tempmap[key_r] = map[key][Math.floor(Math.random() * map[key].length)]
				return tempmap[key_r]
			}

			return map[key][Math.floor(Math.random() * map[key].length)]
		})
		recur_limit -= 1
	}
	return str
}

function showTables(tag, structure, lists) {
	var output = structure + "<br>"

	for (var key in lists){
		if (lists.hasOwnProperty(key)){
			output += "<br/>" + key + "<ol><li>" + lists[key].join("</li><li>") + "</li></ol>"
		}
	}

	document.getElementById(`outputText_${tag}`).innerHTML = output
}

var lists_nss = {
	"Table 1": [`A`, `B`, `C`, `D`],
	"Table 2": [`A`, `<i>B</i>`, `<b>C</b>`, `<u>D</u>`],
	"Table 3": [`A`, `B {Table 1}`, `C {Table 2}`, `D {Table 3}`],
}

function selectRandom_nss() {
	document.getElementById('outputText_nss').innerHTML = expandString(`Result: {Table 1}, {Table 2}, {Table 3}<br><i>Other</i> <b>text</b> <u>goes</u> here<br>Formatting can also apply to table entries: <b>{Table 3}</b>`, lists_nss)
}

function showTables_nss() {
	showTables("nss", "Result: {Table 1}, {Table 2}, {Table 3}<br><i>Other</i> <b>text</b> <u>goes</u> here<br>Formatting can also apply to table entries: <b>{Table 3}</b>", lists_nss)
}

</script>
